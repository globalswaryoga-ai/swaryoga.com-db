# 🚀 FIRST TIME SETUP (Run once)
qa-setup                           # Interactive setup wizard
source ~/.zshrc                    # Reload shell aliases

# 🔌 BRIDGE MANAGEMENT
qa-bridge-status                   # Check if bridge is running
qa-bridge-start                    # Start WhatsApp bridge (Docker)
qa-bridge-restart                  # Restart bridge (fixes most issues)
qa-bridge-stop                     # Stop bridge
qa-bridge-logs                     # View live logs (Ctrl+C to exit)

# 💻 DEVELOPMENT SERVER
qa-dev-start                       # Start Next.js on :3020
qa-qr-open                         # Open QR page in browser

# 📊 DIAGNOSTICS & TESTING
qa-diagnose                        # Full health check (bridges, server, DB)
qa-test-flow                       # Test send/receive pipeline

# 💾 DATABASE QUERIES
qa-db-check                        # Check MongoDB connectivity
qa-db-messages                     # List last 10 messages
qa-db-leads                        # List last 10 leads

# ⚙️  CONFIGURATION
qa-config-bridge                   # Edit bridge .env
qa-config-app                      # Edit app .env.local

# 📖 HELP
qa-help                            # Show all commands (you're reading it!)
